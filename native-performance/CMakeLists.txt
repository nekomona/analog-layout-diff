cmake_minimum_required(VERSION 3.16)
project(layoutdiff_native VERSION 1.0 LANGUAGES CXX)
add_executable(layoutdiff-native src/main.cpp)
target_compile_features(layoutdiff-native PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(layoutdiff-native PRIVATE /W4 /O2)
else()
  target_compile_options(layoutdiff-native PRIVATE -Wall -Wextra -Wpedantic -O3)
endif()
install(TARGETS layoutdiff-native RUNTIME DESTINATION bin)
